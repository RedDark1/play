sw.js (Service Worker)

/* PulseHub Service Worker v3.0 — GitHub Pages Ready */
const CACHE_NAME = 'pulsehub-cache-v3';
const STATIC_ASSETS = [
  './',
  './index.html',
  './manifest.json',
  './icon-192.png',
  './icon-512.png',
  './screenshot-desktop.png'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(STATIC_ASSETS))
  );
  self.skipWaiting();
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(
        keys.filter((key) => key !== CACHE_NAME).map((key) => caches.delete(key))
      )
    )
  );
  self.clients.claim();
});

self.addEventListener('fetch', (event) => {
  if (event.request.method !== 'GET') return;

  event.respondWith(
    fetch(event.request)
      .then((networkResponse) => {
        const url = new URL(event.request.url);
        if (url.origin === self.location.origin && networkResponse.ok) {
          const clone = networkResponse.clone();
          caches.open(CACHE_NAME).then((cache) => cache.put(event.request, clone));
        }
        return networkResponse;
      })
      .catch(() => caches.match(event.request))
  );
});

Пошаговая инструкция упаковки в .APK (Android) и .EXE / .MSIX (Windows)

Бесплатная облачная компиляция нативных пакетов из вашего репозитория GitHub Pages

Открыть PWABuilder.com
01

Загрузка в репозиторий

Нажмите «Скачать весь комплект» выше и поместите все 6 файлов в корень вашего репозитория reddark1/pulsehub на GitHub.

02

Анализ в PWABuilder

Вставьте ссылку https://reddark1.github.io/pulsehub/ в строку анализа PWABuilder. Система проверит манифест, иконки и Service Worker.

03

Генерация .APK и .EXE

Нажмите Package for Stores: выберите Android для получения подписанного .apk / .aab или Windows для .msix / .exe.

Установка PulseHub

Автономный запуск как нативное приложение

Ваш браузер поддерживает мгновенную установку PulseHub на рабочий стол или главный экран:

💻 Windows / macOS (Chrome, Edge, Brave)

Нажмите на значок «Установить PulseHub» в правой части адресной строки браузера или откройте меню ⋮ → Сохранить и поделиться → Установить приложение.

📱 iOS / iPadOS (Safari)

Нажмите кнопку «Поделиться» в нижней панели Safari, пролистайте вниз и выберите пункт «На экран „Домой“».

🤖 Android (Chrome)

Откройте меню браузера (⋮) в правом верхнем углу и нажмите «Установить приложение» или «Добавить на гл. экран».
